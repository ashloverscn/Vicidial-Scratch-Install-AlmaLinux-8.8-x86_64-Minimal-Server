mysql -u root <<EOF
-- Create database
CREATE DATABASE IF NOT EXISTS asterisk CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE asterisk;

-- Core user table
CREATE TABLE IF NOT EXISTS vicidial_users (
    user_id INT(9) NOT NULL AUTO_INCREMENT,
    user VARCHAR(20) NOT NULL,
    pass VARCHAR(100) NOT NULL,
    full_name VARCHAR(50),
    user_level INT(2),
    user_group VARCHAR(20),
    active VARCHAR(1),
    PRIMARY KEY (user_id),
    UNIQUE KEY user (user)
) ENGINE=MyISAM;

-- User login log table
CREATE TABLE IF NOT EXISTS vicidial_user_log (
    user_log_id INT(11) NOT NULL AUTO_INCREMENT,
    user VARCHAR(20),
    event_date DATETIME,
    event VARCHAR(50),
    campaign_id VARCHAR(8),
    computer_ip VARCHAR(15),
    server_ip VARCHAR(15),
    PRIMARY KEY (user_log_id),
    INDEX user_date_idx (user, event_date)
) ENGINE=MyISAM;

-- Server performance table
CREATE TABLE IF NOT EXISTS server_performance (
    start_time DATETIME,
    server_ip VARCHAR(15),
    load_average VARCHAR(10),
    used_ram INT(11),
    total_ram INT(11)
) ENGINE=MyISAM;

-- Live agents table
CREATE TABLE IF NOT EXISTS vicidial_live_agents (
    live_agent_id INT(9) NOT NULL AUTO_INCREMENT,
    user VARCHAR(20),
    server_ip VARCHAR(15),
    conf_exten VARCHAR(20),
    status VARCHAR(6),
    campaign_id VARCHAR(8),
    PRIMARY KEY (live_agent_id)
) ENGINE=MEMORY;
